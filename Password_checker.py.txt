def check_password():
    username=input("Enter username:")
    password=input("Enter password:")
    password2=input("Repeat password:")
    
    special_chars="!@#$%^&*()-=_+[]{{:;'<>?/"
    
    if username.lower() in password.lower() or password.lower() in username.lower() :
        print("usename and password can not contain each other.\n Account can not create.")
    elif len(password)<8 :
        print("password can not be less than 8 charactors.\n Account can not create.") 
    elif password.islower():
         print("password should have at least one capital letter.\n Account can not create.")
    elif password.isupper():
         print("password should have at least one lower letter.\n Account can not create.")
    elif password.isdigit():
         print("password should have at least one letter.\n Account can not create.")
    elif password.isalpha():
         print("password should have at least one number.\n Account can not create.")
    elif not any(char in special_chars for char in password):
         print("password should have at least one special charactors.\n Account can not create.")
    elif password!=password2:
         print("passwords are not match.\n Account can not create.")
         
    else:
        print("Account creat successfully.")
        
check_password()
